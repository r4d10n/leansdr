COMPILATION EXAMPLES:
--------------------

ARM32 with NEON:
  g++ -mfpu=neon -march=armv7-a -Isrc -std=c++11 -O3 myapp.cpp

ARM64 (NEON standard):
  g++ -march=armv8-a -Isrc -std=c++11 -O3 myapp.cpp

x86/x64 (scalar fallback):
  g++ -Isrc -std=c++11 -O3 myapp.cpp

With profiling:
  g++ -DLEANSDR_ENABLE_PROFILING=1 -Isrc -std=c++11 -O3 myapp.cpp

USAGE PATTERN:
--------------

#include "leansdr/config.h"
#include "leansdr/neon_helpers.h"

void my_function(float* out, const float* in, size_t count) {
#if LEANSDR_USE_NEON_INTRINSICS
    if (leansdr::use_neon()) {
        // NEON-optimized path
        process_with_neon(out, in, count);
    } else {
        // Scalar fallback
        process_scalar(out, in, count);
    }
#else
    // Always use scalar on non-ARM
    process_scalar(out, in, count);
#endif
}
